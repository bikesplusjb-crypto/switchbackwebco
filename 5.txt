<script>
// Formspree submission with friendly inline feedback
(function () {
  var form = document.getElementById('contactForm');
  var statusBox = document.getElementById('formStatus');
  if (!form) return;
  form.addEventListener('submit', function (e) {
    e.preventDefault();
    statusBox.className = 'form-status show';
    statusBox.style.color = 'var(--text-2)';
    statusBox.textContent = 'Sending…';
    var data = new FormData(form);
    fetch(form.action, {
      method: 'POST',
      body: data,
      headers: { 'Accept': 'application/json' }
    }).then(function (resp) {
      if (resp.ok) {
        form.reset();
        statusBox.className = 'form-status show success';
        statusBox.textContent = '✓ Got it — we\'ll reply within 24 hours.';
      } else {
        return resp.json().then(function (j) {
          throw new Error(j.error || 'Submission failed');
        });
      }
    }).catch(function (err) {
      statusBox.className = 'form-status show error';
      statusBox.textContent = 'Something went wrong. Please email directly instead — see the address above.';
      console.error(err);
    });
  });
})();
</script>
